import React, { useState } from 'react';
import { ARTWORKS, Artwork } from '../data/artworks';
import { sound } from '../utils/audio';

interface WorkIndexProps {
  onSelectArtwork: (artworkId: string) => void;
}

export const WorkIndex: React.FC<WorkIndexProps> = ({ onSelectArtwork }) => {
  const [hoveredArtwork, setHoveredArtwork] = useState<Artwork>(ARTWORKS[0]);

  return (
    <div className="min-h-screen pt-28 pb-32 px-6 md:px-12 lg:px-16 max-w-[1500px] mx-auto animate-in fade-in duration-300">
      <div className="grid grid-cols-1 lg:grid-cols-12 gap-12 lg:gap-16 items-start">
        {/* Left Column: Title, Subtitle, and Live Interactive Preview Photo */}
        <div className="lg:col-span-5 lg:sticky lg:top-28">
          <h1 className="font-condensed text-7xl sm:text-8xl md:text-[110px] leading-[0.84] text-ink-900 tracking-tight mb-6">
            WORK
          </h1>

          <div className="w-12 h-[1px] bg-neutral-300 mb-6" />

          <p className="text-neutral-500 text-xs sm:text-sm leading-relaxed mb-8 max-w-sm">
            Eleven prints from 2022 — 2025, in five series. Hover a line to lift the print; click it to see the plate.
          </p>

          {/* Large Live Hover Preview Photo */}
          <div className="hidden lg:block relative w-full aspect-[4/3] rounded-xs overflow-hidden bg-[#EBE7DE] p-2.5 shadow-xl transition-all duration-300 group">
            <img
              src={hoveredArtwork.image}
              alt={hoveredArtwork.title}
              className="w-full h-full object-cover transition-opacity duration-300"
            />
            <div className="absolute bottom-4 left-4 right-4 flex items-center justify-between text-white/95 text-[11px] font-mono tracking-wider drop-shadow-md bg-black/30 backdrop-blur-md px-3 py-1.5 rounded-full">
              <span>{hoveredArtwork.number} — {hoveredArtwork.title}</span>
              <span>{hoveredArtwork.place}</span>
            </div>
          </div>
        </div>

        {/* Right Column: Artwork Table */}
        <div className="lg:col-span-7">
          {/* Table Header */}
          <div className="hidden sm:grid grid-cols-12 gap-4 pb-3 mb-2 border-b border-neutral-300/80 text-[11px] font-mono tracking-[0.14em] text-neutral-400 uppercase">
            <div className="col-span-1">NO.</div>
            <div className="col-span-5">TITLE</div>
            <div className="col-span-2">SERIES</div>
            <div className="col-span-3">PLACE</div>
            <div className="col-span-1 text-right">YEAR</div>
          </div>

          {/* Rows */}
          <div className="divide-y divide-neutral-200/80">
            {ARTWORKS.map((item) => {
              const isSelected = hoveredArtwork.id === item.id;
              return (
                <div
                  key={item.id}
                  onMouseEnter={() => {
                    setHoveredArtwork(item);
                    sound.playPaperLift();
                  }}
                  onClick={() => {
                    sound.playTick(600);
                    onSelectArtwork(item.id);
                  }}
                  className={`group py-4 sm:py-4 px-2 sm:px-3 rounded-lg flex items-center sm:grid sm:grid-cols-12 gap-3 sm:gap-4 transition-all duration-200 cursor-pointer ${
                    isSelected ? 'bg-black/[0.04]' : 'hover:bg-black/[0.02]'
                  }`}
                >
                  {/* Thumbnail */}
                  <div className="w-14 sm:w-12 h-10 sm:h-9 flex-shrink-0 rounded-xs overflow-hidden bg-neutral-200 border border-neutral-300/60 shadow-xs">
                    <img
                      src={item.image}
                      alt={item.title}
                      className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-300"
                    />
                  </div>

                  {/* Number */}
                  <div className="hidden sm:block col-span-1 font-mono text-xs text-neutral-400 group-hover:text-neutral-900 transition-colors">
                    {item.number}
                  </div>

                  {/* Title */}
                  <div className="flex-1 sm:col-span-5 font-normal text-sm sm:text-base text-ink-900 group-hover:translate-x-1 transition-transform">
                    {item.title}
                  </div>

                  {/* Series */}
                  <div className="hidden sm:block col-span-2 font-mono text-[11px] tracking-wider text-neutral-400 uppercase">
                    {item.series}
                  </div>

                  {/* Place */}
                  <div className="hidden sm:block col-span-3 font-mono text-[11px] tracking-wider text-neutral-500 uppercase">
                    {item.place}
                  </div>

                  {/* Year */}
                  <div className="sm:col-span-1 font-mono text-xs text-neutral-400 text-right">
                    {item.year}
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      </div>
    </div>
  );
};
